cmake_minimum_required(VERSION 3.28)
project(with_raspberry_pi_cpp_learning LANGUAGES CXX)
include(CTest)
include(CheckCXXSourceCompiles)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
if(NOT CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    message(FATAL_ERROR "The course checker currently supports GCC or Clang driver flags.")
endif()
option(COURSE_REQUIRE_ALL_FEATURES "Fail instead of skipping unsupported library/language examples" OFF)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${PROJECT_BINARY_DIR}")

# These two files are the unchanged CMake examples, with labelled source scaffolding.
add_subdirectory(lessons/part1/1.introduction/cmake-demo)
add_subdirectory(lessons/part4/4.build_systems/cmake-demo)

function(course_program target source standard optional existing)
    set(extra_library "")
    if(optional)
        # Include the actual file: passing its contents through CMake would
        # reinterpret C++ string escapes such as backslash-n in the probe.
        set(probe "#include \"${PROJECT_SOURCE_DIR}/${source}\"\n")
        get_filename_component(probe_dir "${PROJECT_SOURCE_DIR}/${source}" DIRECTORY)
        set(CMAKE_REQUIRED_INCLUDES "${probe_dir}")
        set(CMAKE_CXX_STANDARD "${standard}")
        check_cxx_source_compiles("${probe}" "HAVE_${target}")
        if(NOT HAVE_${target} AND target MATCHES "uncommented_print$")
            # Some libstdc++ releases keep print support in this companion library.
            set(CMAKE_REQUIRED_LIBRARIES stdc++exp)
            check_cxx_source_compiles("${probe}" "HAVE_${target}_EXP")
            if(HAVE_${target}_EXP)
                set(HAVE_${target} TRUE)
                set(extra_library stdc++exp)
            endif()
        endif()
        if(NOT HAVE_${target})
            if(COURSE_REQUIRE_ALL_FEATURES)
                message(FATAL_ERROR "Required example unavailable: ${target}; inspect the compile/link probe log.")
            endif()
            message(STATUS "SKIP ${target}: unavailable with the configured compiler/library; not a passed test")
            if(BUILD_TESTING)
                add_test(NAME ${target}.output COMMAND "${Python3_EXECUTABLE}"
                    "${PROJECT_SOURCE_DIR}/tools/check_program.py" --skip "${target}")
                set_tests_properties(${target}.output PROPERTIES SKIP_RETURN_CODE 77 TIMEOUT 10)
            endif()
            return()
        endif()
    endif()
    if(NOT existing)
        add_executable(${target} "${source}")
    endif()
    set_target_properties(${target} PROPERTIES CXX_STANDARD ${standard} CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS NO)
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
    if(extra_library)
        target_link_libraries(${target} PRIVATE ${extra_library})
    endif()
    if(BUILD_TESTING)
        add_test(NAME ${target}.output COMMAND "${Python3_EXECUTABLE}"
            "${PROJECT_SOURCE_DIR}/tools/check_program.py"
            --target "${target}" --executable "$<TARGET_FILE:${target}>")
        set_tests_properties(${target}.output PROPERTIES TIMEOUT 15)
    endif()
endfunction()

file(READ "${PROJECT_SOURCE_DIR}/programs.json" programs)
string(JSON program_count LENGTH "${programs}")
math(EXPR last "${program_count} - 1")
foreach(index RANGE 0 ${last})
    foreach(key target source standard)
        string(JSON ${key} GET "${programs}" ${index} ${key})
    endforeach()
    string(JSON optional ERROR_VARIABLE optional_missing GET "${programs}" ${index} optional)
    string(JSON existing ERROR_VARIABLE existing_missing GET "${programs}" ${index} existing)
    if(optional_missing)
        set(optional FALSE)
    endif()
    if(existing_missing)
        set(existing FALSE)
    endif()
    course_program("${target}" "${source}" "${standard}" "${optional}" "${existing}")
endforeach()

if(BUILD_TESTING)
    add_subdirectory(tests)
endif()
